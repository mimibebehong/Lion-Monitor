def say(text: str = "") -> None:

def head(title: str) -> None:
    say("")
    say("=" * 92)
    say(title)
    say("=" * 92)


def tone(freq: float, amp: float, seconds: float = 20.0, fs: float = FS, noise: float = 0.0):
    t = np.arange(int(seconds * fs)) / fs
    rng = np.random.default_rng(7)
    x = amp * np.sin(2 * np.pi * freq * t)
    if noise:
        x = x + rng.normal(0.0, noise, x.size)
    return x.astype(np.float32)


# ---------------------------------------------------------------- 1. 参数是否真正被消费
def probe_param_consumption() -> None:
    head("[C1] schema 里声明的参数：哪些真的进了计算（逐个改动、看输出是否变化）")
    x = tone(145.0, 0.8, noise=0.002)
    base = compute_metrics(x, FS)
    say(f"　基准（默认参数）：{ {k: round(v, 6) for k, v in base.items()} }")
    say("")
    say(f"　{'参数':<12}{'改动':<26}{'a_Peak':>14}{'a_RMS':>14}{'v_RMS':>14}  结论")
    trials = [
        ("band_low", {"band_low": 400.0}),
        ("band_high", {"band_high": 300.0}),
        ("detrend", {"detrend": False}),
        ("window_s", {"window_s": 1.0}),
        ("step_s", {"step_s": 1.0}),
        ("window_fn", {"window_fn": "blackman"}),
    ]
    for key, override in trials:
        params = {**default_params("v_RMS"), **override}
        values = compute_metrics(x, FS, params=params)
        changed = any(abs(values[k] - base[k]) > 1e-9 for k in base)
        say(
            f"　{key:<12}{str(override):<26}"
            f"{values['a_Peak']:>14.6f}{values['a_RMS']:>14.6f}{values['v_RMS']:>14.6f}"
            f"  {'生效' if changed else '★ 未消费（输出完全不变）'}"
        )
    # 偏移信号单独验证 detrend 对 a_Peak 的作用
    offset = x + 0.5
    say("")
    say("　加 0.5 g 直流偏置后：")
    for detrend in (True, False):
        values = compute_metrics(offset, FS, params={"detrend": detrend})
        say(
            f"　　detrend={detrend!s:<5} → a_Peak={values['a_Peak']:.6f} g，a_RMS={values['a_RMS']:.6f} g"
        )


# ---------------------------------------------------------------- 2. 带通边界
def probe_band_edges() -> None:
    head("[C2] 带通边界：band_low=0 / band_high>fs/2 时发生什么，参数指纹是否还准")
    x = tone(145.0, 0.8, noise=0.002)
    for params in (
        {"band_low": 0.0, "band_high": 1000.0},
        {"band_low": 0.1, "band_high": 1000.0},
    ):
        try:
            value = compute_metrics(x, FS, params=params, codes=["v_RMS"])["v_RMS"]
            say(f"　params={params} → v_RMS={value:.6f} mm/s")
        except Exception as exc:  # noqa: BLE001
            say(f"　params={params} → 抛出 {type(exc).__name__}: {str(exc).splitlines()[0]}")
        say(f"　　validate_params 结论：{validate_params(params, FS) or '（通过）'}")
    say("")
    high_a = compute_metrics(x, FS, params={"band_low": 10, "band_high": 16000}, codes=["v_RMS"])[
        "v_RMS"
    ]
    high_b = compute_metrics(x, FS, params={"band_low": 10, "band_high": 30000}, codes=["v_RMS"])[
        "v_RMS"
    ]
    say(
        f"　band_high=16000（=fs/2）→ v_RMS={high_a:.9f}；band_high=30000（>fs/2，内部被夹到 15200）→ {high_b:.9f}"
    )
    say(
        f"　两者数值相等：{abs(high_a - high_b) < 1e-12}；但 params_hash 不同："
        f"{params_hash({'band_low': 10, 'band_high': 16000})} vs {params_hash({'band_low': 10, 'band_high': 30000})}"
    )
    say("　→ 指纹只记录「用户填的参数」，不记录「实际用于计算的夹紧后参数」：")


# ---------------------------------------------------------------- 3. 数值边界
def probe_numeric_edges() -> None:
    head("[C3] 数值边界：空 / 常量 / NaN / Inf / 极短 / 超长")
    cases: list[tuple[str, np.ndarray]] = [
        ("空信号（0 点）", np.array([], dtype=np.float32)),
        ("1 点", np.array([1.0], dtype=np.float32)),
        ("10 点（<64）", np.full(10, 0.5, dtype=np.float32)),
        ("63 点（<64）", np.full(63, 0.5, dtype=np.float32)),
        ("64 点（=阈值）", tone(145, 0.8, 64 / FS)),
        ("常量 0（全零）", np.zeros(int(FS * 2), dtype=np.float32)),
        ("常量 5.0 g（恒值）", np.full(int(FS * 2), 5.0, dtype=np.float32)),
        (
            "含 NaN（1 个）",
            np.concatenate([tone(145, 0.8, 2.0), np.array([np.nan], dtype=np.float32)]),
        ),
        ("全 NaN", np.full(int(FS * 2), np.nan, dtype=np.float32)),
        (
            "含 Inf（1 个）",
            np.concatenate([tone(145, 0.8, 2.0), np.array([np.inf], dtype=np.float32)]),
        ),
    ]
    say(f"　{'场景':<20}{'a_Peak':>16}{'a_RMS':>16}{'v_RMS':>16}  结果")
    for label, samples in cases:
        try:
            values = compute_metrics(samples, FS)
            flag = ""
            if any(not np.isfinite(v) for v in values.values()):
                flag = "★ 输出非有限值（NaN/Inf）"
            say(
                f"　{label:<20}{values['a_Peak']:>16.6g}{values['a_RMS']:>16.6g}"
                f"{values['v_RMS']:>16.6g}  {flag}"
            )
        except Exception as exc:  # noqa: BLE001
            say(f"　{label:<20}{'—':>16}{'—':>16}{'—':>16}  抛出 {type(exc).__name__}: {exc}")
    say("")
    say("　判有效性用的 _judge：a_rms <= 1e-9 → 无效；NaN 走不进这个分支（NaN <= 1e-9 为 False）")
    say(
        f"　grade(NaN, '>', 1.5, 3.0) = {grade(float('nan'), '>', 1.5, 3.0)}；"
        f"point_level(NaN, (1.5, 3.0, '>', '指标级')) = {point_level(float('nan'), (1.5, 3.0, '>', '指标级'))}"
    )
    say("　→ NaN 值会被判成「正常」，趋势图上是一条断线/空白，健康评级仍是绿色。")

# ---------------------------------------------------------------- 5. 混叠
def probe_aliasing() -> None:
    head("[C5] 32 kHz 采样下的 FFT：软件侧有没有抗混叠 / 降采样")
    fs = 32000.0
    seconds = 2.0
    t = np.arange(int(seconds * fs)) / fs
    for freq in (8000.0, 12000.0, 20000.0, 30000.0):
        x = np.sin(2 * np.pi * freq * t).astype(np.float32)
        spec = spectrum_from_samples(x, fs, label=f"{freq:g} Hz 输入")
        peak = spec.main_peak()
        alias = abs(freq - fs) if freq > fs / 2 else freq
        say(
            f"　输入 {freq:>7.0f} Hz → 谱峰 {peak.freq_hz if peak else float('nan'):>7.2f} Hz"
            f"（理论折叠位置 {alias:>7.0f} Hz）"
        )
    say(
        "　→ 软件全程用原始采样率做 rfft、不做降采样（只有上屏抽样用 min/max 包络），"
        "所以不会「软件再引入」混叠；但也没有任何抗混叠检测：>fs/2 的输入会按物理折叠出现在谱上，"
        "页面照常显示为真实频率。"
    )


# ---------------------------------------------------------------- 6. v_RMS 隐藏口径
def probe_vrms_edges() -> None:
    head("[C6] v_RMS 的隐藏口径：0.5 s 首尾被切掉（边缘暂态处理）")
    x = tone(145.0, 0.8, 20.0, noise=0.002)
    trimmed = compute_metrics(x, FS, codes=["v_RMS"])["v_RMS"]
    # 复现 _velocity_rms 的算法但不切边
    from lion_monitor.features.metrics import _band_filter, _integrate_to_velocity

    band = (10.0, 1000.0)
    filtered = _band_filter(x - x.mean(), FS, band)
    full = float(np.sqrt(np.mean(_integrate_to_velocity(filtered, FS, band) ** 2)))
    say(
        f"　切掉首尾各 0.5 s（现口径）v_RMS={trimmed:.6f} mm/s；全程不切 v_RMS={full:.6f} mm/s；"
        f"差 {(trimmed / full - 1) * 100:+.3f}%"
    )
    say(
        f"　20 s 记录里实际参与计算的时长 = 20 - 1 = 19 s（{19 / 20:.1%}）；方案/页面均未写这条口径"
    )

def probe_dedupe_scope() -> None:
    head("[C7] params_hash 去重：cache 是否跨通道复用（同一批 4 通道同参数）")
    WORK.mkdir(parents=True, exist_ok=True)
    db = WORK / "review-dedupe.db"
    if db.exists():
        db.unlink()
    engine = make_engine(db)
    init_db(engine)
    sess = make_session_factory(engine)()
    task = tasks.create_task(sess, name="去重", operator="review")
    obj = object_registry.create_object(sess, task.id, "轴承", "轴承", part="X")
    metric_def.seed_defaults(sess, task.id)
    root = tree.root_node(sess, task.id)
    dev = tree.create_node(sess, task.id, "device", "线", parent_id=root.id, attrs={})
    comp = tree.create_node(
        sess, task.id, "component", "轴承", parent_id=dev.id, attrs={}, ref_id=obj.id
    )
    for code in ("CH1", "CH2", "CH3", "CH4"):
        ch = tree.create_node(sess, task.id, "channel", code, parent_id=comp.id, attrs={})
        for metric in ("a_Peak", "a_RMS", "v_RMS"):
            tree.create_node(
                sess, task.id, "metric", metric, parent_id=ch.id, attrs={"code": metric}
            )
            metric_assign.assign_metric(sess, task.id, ch.id, metric)
    batch = WORK / "batch-4ch"
    if batch.exists():
        for p in batch.glob("*.dat"):
            p.unlink()
    for code, amp in (("CH1", 0.1), ("CH2", 0.8), ("CH3", 0.2), ("CH4", 0.3)):
        write_dat(
            batch / f"CH_{code}.dat",
            samples=tone(145, amp, 2.0),
            channel=code,
            sample_rate_hz=FS,
            started_at="20261031120000",
        )
    result = import_batch(sess, task.id, batch, {}, compute=True)
    say(
        f"　4 通道同参数导入：真算 {result.compute_runs} 次 / 复用 {result.dedupe_hits} 次 "
        f"（若跨通道复用会是 3 次）"
    )
    say(
        f"　各通道 a_RMS：{[round(r.value, 4) for r in metric_store.list_values(sess, task.id, 'CH2', 'a_RMS')]}"
    )
    say(
        "　→ 去重 cache 在 pipeline.import_batch 里按文件新建（pipeline.py:361），"
        "key 只有 (指标码, 参数指纹, 转速)，不含通道/数据集：跨通道不会误复用，"
        "但一旦有人把 cache 提到循环外就会串值（属于易踩的结构性风险，不是当前缺陷）。"
    )
    sess.close()


# ---------------------------------------------------------------- 8. 事务与半批数据
def probe_transaction(partial: bool) -> None:
    title = (
        "[C8] 计算阶段抛异常（节点参数 band_low=0）：批次留下多少半成品"
        if partial
        else "[C9] 4 通道里 1 个文件损坏：数据集状态与其余通道"
    )
    head(title)
    WORK.mkdir(parents=True, exist_ok=True)
    db = WORK / ("review-tx-bad.db" if partial else "review-tx-one.db")
    if db.exists():
        db.unlink()
    engine = make_engine(db)
    init_db(engine)
    sess = make_session_factory(engine)()
    task = tasks.create_task(sess, name="事务", operator="review")
    obj = object_registry.create_object(sess, task.id, "轴承", "轴承", part="X")
    metric_def.seed_defaults(sess, task.id)
    root = tree.root_node(sess, task.id)
    dev = tree.create_node(sess, task.id, "device", "线", parent_id=root.id, attrs={})
    comp = tree.create_node(
        sess, task.id, "component", "轴承", parent_id=dev.id, attrs={}, ref_id=obj.id
    )
    ch_ids = {}
    for code in ("CH1", "CH2", "CH3", "CH4"):
        ch = tree.create_node(sess, task.id, "channel", code, parent_id=comp.id, attrs={})
        ch_ids[code] = ch.id
        for metric in ("a_Peak", "a_RMS", "v_RMS"):
            tree.create_node(
                sess, task.id, "metric", metric, parent_id=ch.id, attrs={"code": metric}
            )
            params = {"band_low": 0.0, "band_high": 1000.0} if (partial and code == "CH3") else None
            metric_assign.assign_metric(sess, task.id, ch.id, metric, params=params)
    batch = WORK / ("batch-tx-bad" if partial else "batch-tx-one")
    if batch.exists():
        for p in batch.glob("*.dat"):
            p.unlink()
    for code in ("CH1", "CH2", "CH3", "CH4"):
        write_dat(
            batch / f"CH_{code}.dat",
            samples=tone(145, float(code[-1]) * 0.2, 2.0),
            channel=code,
            sample_rate_hz=FS,
            started_at="20261031121000",
        )
    if not partial:
        # 把 CH4 截断（头部声明点数与实际长度不符）
        path = batch / "CH_CH4.dat"
        raw = path.read_bytes()
        path.write_bytes(raw[:-4096])

    before = len(datasets.list_datasets(sess, task.id))
    try:
        result = import_batch(sess, task.id, batch, {}, compute=True)
        say(
            f"　导入返回：status={result.status} files_ok={result.files_ok}/{result.channels} "
            f"指标值 {result.metrics_written} 行 · 坏文件 {[(b.filename, b.reason) for b in result.files_bad]}"
        )
    except Exception as exc:  # noqa: BLE001
        sess.rollback()
        say(f"　导入抛出 {type(exc).__name__}: {str(exc).splitlines()[0]}")
        say(traceback.format_exc().splitlines()[-3].strip())
    say(f"　数据集数 导入前 {before} → 导入后 {len(datasets.list_datasets(sess, task.id))}")
    with engine.connect() as conn:
        rows = [
            tuple(r)
            for r in conn.exec_driver_sql(
                "SELECT d.code, d.validity, c.code, c.validity, c.validity_note FROM dataset d "
                "LEFT JOIN channel c ON c.dataset_id=d.id ORDER BY c.code"
            )
        ]
        values = conn.exec_driver_sql(
            "SELECT COUNT(*) FROM metric_value WHERE deleted_at IS NULL"
        ).scalar()
        per_ch = [
            tuple(r)
            for r in conn.exec_driver_sql(
                "SELECT ch.code, COUNT(mv.id) FROM channel ch LEFT JOIN metric_value mv ON mv.channel_id=ch.id "
                "GROUP BY ch.code ORDER BY ch.code"
            )
        ]
    say("　库内 dataset/channel 状态：")
    for row in rows:
        say(f"　　{row}")
    say(f"　有效指标值行数 {values}；按通道：{per_ch}")
    sess.close()

    ch = tree.create_node(sess, task.id, "channel", "CH1", parent_id=comp.id, attrs={})

    batch = WORK / "batch-nan"

            p.unlink()

    write_dat(
        batch / "CH_CH1.dat",
        samples=x,
        channel="CH1",
        sample_rate_hz=FS,
        started_at="20261031122000",
    )
    try:
        result = import_batch(sess, task.id, batch, {}, compute=True)
        say(
            f"　导入：status={result.status} files_ok={result.files_ok} 指标值 {result.metrics_written} 行 "

        )

            f"　　{row.metric_code}={row.value!r} unit={row.unit} 判定={point_level(row.value, (1.5, 3.0, '>', 'x'))}"
        )
    health = node_health(sess, task.id)
    for node_id, item in health.items():
        node_row = next(n for n in tree.list_nodes(sess, task.id) if n.id == node_id)
        say(
            f"　　健康 {node_row.type_text} {node_row.name}: {item.state} latest={item.latest_value!r} 超限 {item.over_count}"
        )
    sess.close()
