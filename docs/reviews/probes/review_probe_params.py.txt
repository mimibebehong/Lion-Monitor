OUT = ROOT / "reports" / "2026-10-31-review-params.txt"

lines: list[str] = []


def say(text: str = "") -> None:
    print(text)
    lines.append(text)


def head(title: str) -> None:
    say("")
    say("=" * 92)
    say(title)
    say("=" * 92)


def fresh(name: str):
    WORK.mkdir(parents=True, exist_ok=True)
    path = WORK / name
    if path.exists():
        path.unlink()
    engine = make_engine(path)
    init_db(engine)
    return path, engine, make_session_factory(engine)()


def build(
    session,
    name: str,
    default_high: float = 1000.0,
    override: float | None = None,
    locked: bool = False,
):
    task = tasks.create_task(session, name=name, operator="review")
    obj = object_registry.create_object(session, task.id, "轴承", "轴承", part="X")
    metric_def.seed_defaults(session, task.id)
    # 改指标定义默认（模拟「另一个任务把定义默认改成 500」）
    definition = metric_def.find_metric(session, task.id, "v_RMS")
    defaults = dict(definition.defaults)
    defaults["band_high"] = default_high
    metric_def.update_metric(session, definition.id, defaults=defaults)
    root = tree.root_node(session, task.id)
    dev = tree.create_node(session, task.id, "device", "线", parent_id=root.id, attrs={})
    comp = tree.create_node(
        session, task.id, "component", "轴承", parent_id=dev.id, attrs={}, ref_id=obj.id
    )
    ch_row = channel_reg.create_channel(session, task.id, obj.id, "CH1")
    ch = tree.create_node(
        session, task.id, "channel", "CH1", parent_id=comp.id, attrs={}, ref_id=ch_row.id
    )
    metric_row = metric_def.find_metric(session, task.id, "v_RMS")
    node = tree.create_node(
        session,
        task.id,
        "metric",
        "v_RMS",
        parent_id=ch.id,
        attrs={"code": "v_RMS"},
        ref_id=metric_row.id if metric_row else None,
    )
    params = {"band_high": override} if override is not None else None
    row = metric_assign.assign_metric(
        session, task.id, ch.id, "v_RMS", params=params, params_locked=locked
    )
    return task, ch, node, row


# ---------------------------------------------------------------- 1. 三处一致性
def probe_three_places() -> None:
    head("[T1] 参数三级继承：导入计算 / 页面展示 / 模板导出 三处是否一致")
    path, engine, sess = fresh("params-three.db")
    task_a, ch_a, node_a, assign_a = build(sess, "源任务", default_high=1000.0, override=2000.0)
    task_b, ch_b, node_b, assign_b = build(sess, "目标任务", default_high=500.0)
    task_c, ch_c, node_c, assign_c = build(sess, "只跟默认", default_high=1000.0)

    jobs, _ = metric_jobs(sess, task_a.id)
    job = next(j for j in jobs if j.code == "v_RMS")
    say("　A 节点（定义默认 1000，节点覆盖 2000）：")
    say(
        f"　　导入计算（metric_jobs）生效参数 band_high={job.params['band_high']}　指纹 {job.params_hash}"
    )
    say(
        f"　　页面展示（effective_params）: {metric_assign.effective_params(sess, assign_a.id)['band_high']}"
    )
    payload = tree.export_template(sess, task_a.id)
    exported = None

    def walk(node, path_acc=()):
        nonlocal exported
        if node.get("type") == "metric" and node.get("metric") == "v_RMS":
            exported = node
        for child in node.get("children") or []:
            walk(child, (*path_acc, child["name"]))

    walk(payload["root"])
    say(
        f"　　模板导出：params={exported.get('params')} locked={exported.get('locked')} "
        f"（导出的是节点存储的覆盖项，不是生效参数）"
    )

    jobs_c, _ = metric_jobs(sess, task_c.id)
    job_c = next(j for j in jobs_c if j.code == "v_RMS")
    exported_c = None

    def walk_c(node):
        nonlocal exported_c
        if node.get("type") == "metric" and node.get("metric") == "v_RMS":
            exported_c = node
        for child in node.get("children") or []:
            walk_c(child)

    payload_c = tree.export_template(sess, task_c.id)
    walk_c(payload_c["root"])
    say("")
    say(
        f"　C 节点（定义默认 1000，无覆盖）：导入计算 band_high={job_c.params['band_high']}；"
        f"模板导出 params={exported_c.get('params')}（空）"
    )

    preview = tree.preview_import(sess, task_b.id, payload_c)
    report = tree.import_template(sess, task_b.id, payload_c)
    jobs_b, _ = metric_jobs(sess, task_b.id)
    job_b = next(j for j in jobs_b if j.code == "v_RMS")
    say("　把 C 的模板导入 B（B 的定义默认是 500）：")
    say(
        f"　　导入前 B 的 band_high 应为 500；导入后 B 节点生效参数 band_high={job_b.params['band_high']}"
        f"　→ {'★ 由 1000 静默变成 500（覆盖项为空，跟了目标任务的默认）' if job_b.params['band_high'] == 500.0 else '与源任务一致'}"
    )
    say(
        f"　　preview：树节点 {preview.node_total} · 绑定 {preview.metric_bindings} · 未匹配 {len(preview.unmatched)}"
    )
    say(
        "　结论：导入计算与页面展示都走 compose_params / effective_params（同一函数），两者一致；"
        "模板导出只带「覆盖项 diff」，不带定义默认，跨任务导入后默认值不同的指标会静默漂移。"
    )
    sess.close()


# ---------------------------------------------------------------- 2. params_hash
def probe_hash() -> None:
    head("[T2] params_hash：输入内容与去重边界")
    base = {
        "band_low": 10,
        "band_high": 1000,
        "window_s": 2.0,
        "step_s": 2.0,
        "window_fn": "hanning",
        "detrend": True,
    }
    reordered = {
        "detrend": True,
        "window_fn": "hanning",
        "step_s": 2.0,
        "window_s": 2.0,
        "band_high": 1000,
        "band_low": 10,
    }
    say(f"　默认参数指纹：{params_hash(base)}")
    say(
        f"　同内容不同键序：{params_hash(reordered)}　→ 键排序后稳定：{params_hash(base) == params_hash(reordered)}"
    )
    say(
        f"　1000 与 1000.0：{params_hash({'band_high': 1000})} vs {params_hash({'band_high': 1000.0})}"
    )
    say(
        "　→ hash 输入只有参数 JSON：不含指标码、通道号、数据集、采样率；"
        "去重靠 pipeline 里「每文件一个 cache」限定作用域（pipeline.py:361）"
    )
    path, engine, sess = fresh("params-hash.db")
    task_a, *_ = build(sess, "A", default_high=1000.0, override=2000.0)
    task_b, *_ = build(sess, "B", default_high=1000.0, override=2000.0)
    jobs, _ = metric_jobs(sess, task_a.id)
    say(
        f"　两个不同任务、同通道名、同参数 → v_RMS 指纹都是 {jobs[0].params_hash if jobs else '—'}"
        "（跨任务同值是可接受的：指纹只描述参数）"
    )
    sess.close()


# ---------------------------------------------------------------- 3. 判级调用点
def probe_grading_sites() -> None:
    head("[T3] 阈值判级：还有谁在自己判级（不走 limits_rules.grade）")
    say("　走 limits_rules.grade 的（单一来源）：")
    for item in (
        "features/health.py:48 point_level → grade（趋势着色 / 健康评级 / 概览都用它）",
        "features/overview.py:143 通过 point_level 判级",
        "app/pages/trends.py:217/262/312 通过 point_level",
        "store/limits.py set_limit → limits_rules.validate_limit",
        "store/tree_template.py:392/696 → validate_limit",
        "app/pages/thresholds.py:249 → validate_limit",
    ):
        say(f"　　· {item}")
    say("")
    say("　仍在自己判级的（实测调用方）：")
    say("　　① config.ChannelConfig.classify（config.py:61-71，warn_low/high + alert_low/high）")
    say(
        "　　② configs/example_line.yaml 用的 crit_low/crit_high —— ChannelConfig 里根本没有这两个字段"
    )
    line = config.load_line_config(ROOT / "configs" / "example_line.yaml")
    channel = line.station("ST-COATING").channel("areal_density")
    say(
        f"　　　实测 parse 结果：warn_low={channel.warn_low} warn_high={channel.warn_high} "
        f"alert_low={channel.alert_low} alert_high={channel.alert_high}"
        "（crit_* 被 pydantic 静默丢弃，classify 永远不会出 alert）"
    )
    say("　　　load_line_config 的调用方数量：0（grep 全仓仅定义与 __all__）")
    say("　　③ store/thresholds.py 的 DEFAULT_THRESHOLDS / resolve_threshold（旧 Threshold 表）")
    say("　　　resolve_threshold 调用方数量：0；但 tasks.create_task 仍播种：")
    path, engine, sess = fresh("params-legacy.db")
    task = tasks.create_task(sess, name="旧阈值", operator="review")
    with engine.connect() as conn:
        rows = [
            tuple(r)
            for r in conn.exec_driver_sql(
                "SELECT scope_type,scope_code,metric_code,warn,alert,direction FROM threshold"
            )
        ]
    say(f"　　　新任务自动写入 threshold 表 {len(rows)} 行：{rows}")
    say("　　　→ 这是一份「看起来像生效阈值、实际没人读」的副本；metric_limit 才是真源")
    sess.close()

    say("")
    say("　双向阈值的判级实跑：")
    for direction, warn, alert in ((">", 1.5, 3.0), ("<", 3.0, 1.5)):
        values = [0.5, 2.0, 4.0]
        levels = [limits_rules.grade(v, direction, warn, alert) for v in values]
        say(
            f"　　direction={direction} warn={warn} alert={alert} → {dict(zip(values, levels, strict=True))}"
        )
    say(
        f"　　NaN：grade(NaN,'>',1.5,3.0)={limits_rules.grade(float('nan'), '>', 1.5, 3.0)}，"
        f"point_level(NaN, (1.5,3.0,'>','x'))={point_level(float('nan'), (1.5, 3.0, '>', 'x'))}"
    )


# ---------------------------------------------------------------- 4. 转速
def probe_rpm_and_stale() -> None:
    head("[T4] 转速取值范围与 stale 的实际可见性")
    path, engine, sess = fresh("params-rpm.db")
    task = tasks.create_task(sess, name="转速", operator="review")
    obj = object_registry.create_object(sess, task.id, "轴承", "轴承", part="X")
    metric_def.seed_defaults(sess, task.id)
    root = tree.root_node(sess, task.id)
    dev = tree.create_node(sess, task.id, "device", "线", parent_id=root.id, attrs={"转速": "1500"})
    comp = tree.create_node(
        sess, task.id, "component", "轴承", parent_id=dev.id, attrs={}, ref_id=obj.id
    )
    ch = tree.create_node(sess, task.id, "channel", "CH1", parent_id=comp.id, attrs={})
    nodes = tree.list_nodes(sess, task.id)
    say(f"　转速只写在「设备」节点（通道上 2 级）：node_rpm(通道) = {tree.node_rpm(nodes, ch.id)}")
    say(
        "　转速写在「任务」根节点（通道上 3 级，超出 depth=3）：",
    )
    tree.update_node(sess, root.id, attrs={"转速": "1480"})
    nodes = tree.list_nodes(sess, task.id)
    say(f"　　node_rpm(通道) = {tree.node_rpm(nodes, ch.id)}（根节点的值取不到）")
    say(
        "　→ node_rpm 默认 depth=3，从通道往上只找 通道→对象→设备，写在任务根节点的转速被忽略；"
        "方案 3.2 只说「节点手填」，没写查找深度限制"
    )

    metric_assign.assign_metric(sess, task.id, ch.id, "a_RMS", params={"detrend": False})
    rows = metric_assign.list_assignments(sess, task.id)
    target = next(r for r in rows if r.code == "a_RMS")
    from lion_monitor.store import datasets
    from lion_monitor.store import metrics as metric_store

    ds_row = datasets.create_dataset(sess, task.id, "b1", "/tmp/x", "2026-10-31 09:00:00", "有效")
    file_row = datasets.add_file(sess, ds_row.id, "/tmp/x/CH1.dat", "CH1", 32000.0, 640000)
    metric_store.upsert_node_values(
        sess,
        file_row.id,
        "2026-10-31 09:00:00",
        [
            metric_store.NodeValue(
                node_id=ch.id,
                code="a_RMS",
                value=0.5,
                unit="g",
                params_hash="old",
                rpm_used=None,
                rpm_source="",
            )
        ],
    )
    metric_assign.update_params(sess, target.id, {"detrend": True})
    values = metric_store.list_values(sess, task.id, "CH1", "a_RMS")
    say(
        f"　改参数后库里行（值, stale, 指纹）：{[(r.value, r.stale, r.params_hash) for r in values]}"
    )
    say(
        "　→ stale 只有库字段，没有任何页面读取（grep app/ 只有 metric_params_ui 的注释），"
        "趋势页照常把 stale 值画成正常点"
    )
    sess.close()


def main() -> None:
    head(f"Lion-Monitor 总体复查 · 参数/阈值探测　{datetime.now():%Y-%m-%d %H:%M:%S}")
    probe_three_places()
    probe_hash()
    probe_grading_sites()
    probe_rpm_and_stale()
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text("\n".join(lines) + "\n", encoding="utf-8")
    print(f"\n已写：{OUT}")


if __name__ == "__main__":
    main()
