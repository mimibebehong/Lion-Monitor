"""总体复查·只读探测（大数据量下的页面取数耗时：overview / health / trend）。

复用 review_probe_perf.py 造好的临时库（~/workspace/temp/<日期>/review-perf/perf-long.db，
24 万行指标值），只读不写。运行：
    cd ~/workspace/projects/Lion-Monitor && .venv/bin/python scripts/review_probe_scale.py
"""

from __future__ import annotations

import sys
import time
from datetime import datetime
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))

from lion_monitor.features.health import node_health  # noqa: E402
from lion_monitor.features.overview import task_overview  # noqa: E402
from lion_monitor.store import datasets, tasks, tree  # noqa: E402
from lion_monitor.store import metrics as metric_store  # noqa: E402
from lion_monitor.store.db import make_engine, make_session_factory  # noqa: E402

STAMP = datetime.now().strftime("%Y-%m-%d")
WORK = Path.home() / "workspace" / "temp" / STAMP / "review-perf"
OUT = ROOT / "reports" / "2026-10-31-review-scale.txt"

lines: list[str] = []


def say(text: str = "") -> None:
    print(text)
    lines.append(text)


def timed(label: str, fn):
    start = time.perf_counter()
    value = fn()
    say(f"　{label:<48}{(time.perf_counter() - start) * 1000:>10.1f} ms")
    return value


def main() -> None:
    say("=" * 92)
    say(f"Lion-Monitor 总体复查 · 大数据量取数耗时　{datetime.now():%Y-%m-%d %H:%M:%S}")
    say("=" * 92)
    for name in ("perf-long.db", "perf-many.db", "perf-small.db"):
        path = WORK / name
        if not path.exists():
            say(f"　缺文件：{path}（先跑 review_probe_perf.py）")
            continue
        engine = make_engine(path)
        sess = make_session_factory(engine)()
        task = tasks.list_tasks(sess)[-1]
        with engine.connect() as conn:
            rows = conn.exec_driver_sql("SELECT COUNT(*) FROM metric_value").scalar()
            nodes = conn.exec_driver_sql("SELECT COUNT(*) FROM task_node").scalar()
        say("")
        say(f"　库 {name}：任务 {task.code}　节点 {nodes} 个　metric_value {rows} 行")
        timed("datasets.list_datasets", lambda: datasets.list_datasets(sess, task.id))
        timed(
            "metrics.list_values(CH2, v_RMS)",
            lambda: metric_store.list_values(sess, task.id, "CH2", "v_RMS"),
        )
        timed("health.node_health", lambda: node_health(sess, task.id))
        timed(
            "overview.task_overview（工作台 + 九步 + 超限逐点）",
            lambda: task_overview(sess, task.id),
        )
        overview = task_overview(sess, task.id)
        say(
            f"　　概览产出：超限记录 {len(overview.over_records)} 条，健康分布 {overview.health_counts}"
        )
        timed("tree.tree_snapshot", lambda: tree.tree_snapshot(sess, task.id))
        sess.close()
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text("\n".join(lines) + "\n", encoding="utf-8")
    print(f"\n已写：{OUT}")


if __name__ == "__main__":
    main()
