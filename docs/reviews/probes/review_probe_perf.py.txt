def say(text: str = "") -> None:
    print(text)
    lines.append(text)


def head(title: str) -> None:
    say("")
    say("=" * 92)
    say(title)
    say("=" * 92)


def ms(start: float) -> str:
    return f"{(time.perf_counter() - start) * 1000:.1f} ms"


def timed(label: str, fn):
    start = time.perf_counter()
    value = fn()
    say(f"　{label:<44}{ms(start):>12}")
    return value


def build_tree(session, task_name: str, channels=("CH1", "CH2", "CH3", "CH4")):
    task = tasks.create_task(session, name=task_name, operator="review")
    obj = object_registry.create_object(session, task.id, "隔膜过辊轴承", "轴承", part="隔膜过辊")
    metric_def.seed_defaults(session, task.id)
    defs = {row.code: row for row in metric_def.list_metrics(session, task.id)}
    root = tree.root_node(session, task.id)
    dev = tree.create_node(session, task.id, "device", "卷绕产线", parent_id=root.id, attrs={})
    comp = tree.create_node(
        session, task.id, "component", "隔膜过辊轴承", parent_id=dev.id, attrs={}, ref_id=obj.id
    )
    ch_ids = {}
    for code in channels:
        ch = tree.create_node(session, task.id, "channel", code, parent_id=comp.id, attrs={})
        ch_ids[code] = ch.id
        for metric in ("a_Peak", "a_RMS", "v_RMS"):
            tree.create_node(
                session,
                task.id,
                "metric",
                metric,
                parent_id=ch.id,
                attrs={"code": metric, "单位": defs[metric].unit},
                ref_id=defs[metric].id,
            )
            metric_assign.assign_metric(session, task.id, ch.id, metric)
    return task, ch_ids


def fresh(name: str):
    WORK.mkdir(parents=True, exist_ok=True)
    path = WORK / name
    if path.exists():
        path.unlink()
    engine = make_engine(path)
    init_db(engine)
    return path, engine, make_session_factory(engine)()


# ---------------------------------------------------------------- 1. 现场单批导入实测
def probe_single_import() -> None:
    head("[F1] 现场单批次导入实测：4 通道 × 640 000 点（20 s @32 kHz）")
    path, engine, sess = fresh("perf-single.db")
    task, ch_ids = build_tree(sess, "单批实测")
    batch = sorted(p for p in DATA.iterdir() if p.is_dir())[0]
    files = list(batch.glob("*.dat"))
    say(
        f"　数据：{batch}　{len(files)} 个文件　"
        f"{sum(f.stat().st_size for f in files) / 2**20:.1f} MiB"
    )

    def scan():
        return scan_vibration_dir(batch)

    metas = timed("① 扫描（只读 512 B 头）", scan)
    jobs, skipped = timed("② 解析计算任务（metric_jobs）", lambda: metric_jobs(sess, task.id))
    say(f"　　计算任务 {len(jobs)} 个（每通道 3 指标 × 4 通道），跳过 {len(skipped)} 个")
    say(
        f"　　计划：{[(row.channel_code, row.metric_codes) for row in metric_plan(sess, task.id)][:2]} …"
    )

    def one_read():
        return read_vibration(metas[0].path)

    record = timed("③ 读一个 20 s 文件（含 scale 换算）", one_read)
    say(
        f"　　样本 {record.samples.size} 点 dtype={record.samples.dtype} "
        f"常驻 {record.samples.nbytes / 2**20:.1f} MiB"
    )

    peak_before = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    tracemalloc.start()
    t0 = time.perf_counter()
    result = import_batch(sess, task.id, batch, ch_ids, compute=True)
    elapsed = time.perf_counter() - t0
    current, peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()
    peak_after = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    say(
        f"　④ import_batch（结构 + 计算 + 落库）：{elapsed * 1000:.0f} ms；"
        f"真算 {result.compute_runs} 次 / 复用 {result.dedupe_hits} 次 / 指标值 {result.metrics_written} 行"
    )
    say(
        f"　　Python 分配峰值 {peak / 2**20:.1f} MiB（tracemalloc）；进程 RSS 峰值 "
        f"{peak_after / 1024:.1f} MiB（导入前 {peak_before / 1024:.1f} MiB，增量 "
        f"{(peak_after - peak_before) / 1024:.1f} MiB）"
    )
    say(
        f"　　峰值内存量级说明：4 个通道逐个 read_vibration（每个 {record.samples.nbytes / 2**20:.1f} MiB float32），"
        "同一时刻只有当前通道样本常驻；落库是每文件一次 commit（无整批事务）"
    )
    sess.close()


# ---------------------------------------------------------------- 2. 100 批次容量
def probe_many_batches(count: int = 100) -> None:
    head(f"[F2] 连续导入 {count} 个批次后：趋势 / 健康 / 数据集页查询是否退化")
    path, engine, sess = fresh("perf-many.db")
    task, ch_ids = build_tree(sess, "多批次")
    batch_root = WORK / f"batches-{count}"
    if not batch_root.exists():
        batch_root.mkdir(parents=True, exist_ok=True)
        rng = np.random.default_rng(11)
        for index in range(count):
            sub = batch_root / f"batch{index:03d}"
            sub.mkdir(exist_ok=True)
            started = f"202610{1 + index // 24:02d}{index % 24:02d}0000"
            for code in ("CH1", "CH2", "CH3", "CH4"):
                t = np.arange(int(2 * FS)) / FS
                samples = (
                    0.5 * np.sin(2 * np.pi * (145 + index) * t) + rng.normal(0, 0.01, t.size)
                ).astype(np.float32)
                write_dat(
                    sub / f"CH_{code}.dat",
                    samples=samples,
                    channel=code,
                    sample_rate_hz=FS,
                    started_at=started,
                )

    times = []
    t_all = time.perf_counter()
    for index in range(count):
        start = time.perf_counter()
        import_batch(sess, task.id, batch_root / f"batch{index:03d}", ch_ids, compute=True)
        times.append(time.perf_counter() - start)
    total = time.perf_counter() - t_all
    say(
        f"　{count} 批导入合计 {total:.1f} s，单批 最小 {min(times) * 1000:.0f} / 中位 "
        f"{sorted(times)[len(times) // 2] * 1000:.0f} / 最大 {max(times) * 1000:.0f} ms"
    )
    with engine.connect() as conn:
        for table in ("dataset", "channel", "metric_value"):
            n = conn.exec_driver_sql(f"SELECT COUNT(*) FROM {table}").scalar()
            say(f"　　{table} 行数 {n}")

    say("")
    say("　查询耗时（同一任务、同一树结构，rows 随批次数增长）：")
    timed(
        "datasets.list_datasets（含每批 list_files + stat）",
        lambda: datasets.list_datasets(sess, task.id),
    )
    timed(
        "metrics.list_values(CH2, v_RMS)",
        lambda: metric_store.list_values(sess, task.id, "CH2", "v_RMS"),
    )
    timed(
        "metrics.list_values(CH2, 全部指标 3 次)",
        lambda: [
            metric_store.list_values(sess, task.id, "CH2", c) for c in ("a_Peak", "a_RMS", "v_RMS")
        ],
    )
    timed("health.node_health（整棵树评级）", lambda: node_health(sess, task.id))
    timed("overview.task_overview（工作台）", lambda: task_overview(sess, task.id))

    # 与 3 批次基线对照：另建一个小库
    say("")
    say("　基线对照（3 批次，同样树结构）：")
    path2, engine2, sess2 = fresh("perf-small.db")
    task2, ch2 = build_tree(sess2, "少批次")
    for index in range(3):
        import_batch(sess2, task2.id, batch_root / f"batch{index:03d}", ch2, compute=True)
    timed("datasets.list_datasets", lambda: datasets.list_datasets(sess2, task2.id))
    timed(
        "metrics.list_values(CH2, v_RMS)",
        lambda: metric_store.list_values(sess2, task2.id, "CH2", "v_RMS"),
    )
    timed("health.node_health", lambda: node_health(sess2, task2.id))
    timed("overview.task_overview", lambda: task_overview(sess2, task2.id))
    sess2.close()
    sess.close()


# ---------------------------------------------------------------- 3. 长序列绘图
def probe_long_series() -> None:
    head("[F3] 长序列：一次算 20 s → S4 切窗后单批点数会放大，页面是否分页/降采样")
    path, engine, sess = fresh("perf-long.db")
    task, ch_ids = build_tree(sess, "长序列")
    ds_row = datasets.create_dataset(
        sess, task.id, "长序列", "/tmp/long", "2026-10-31 09:00:00", "有效"
    )
    rows_per_channel = 20000
    values = 0
    from sqlalchemy import text

    base = datetime(2026, 1, 1)
    for code in ("CH1", "CH2", "CH3", "CH4"):
        f_row = datasets.add_file(sess, ds_row.id, f"/tmp/long/{code}.dat", code, FS, 640000)
        for metric in ("a_Peak", "a_RMS", "v_RMS"):
            node = next(
                n
                for n in tree.list_nodes(sess, task.id)
                if n.node_type == "metric" and n.name == metric and n.parent_id == ch_ids[code]
            )
            payload = [
                {
                    "c": f_row.id,
                    "n": node.id,
                    "code": metric,
                    "v": float(index % 100) / 10,
                    "t": (base + __import__("datetime").timedelta(seconds=index * 2)).strftime(
                        "%Y-%m-%d %H:%M:%S"
                    ),
                }
                for index in range(rows_per_channel)
            ]
            sess.execute(
                text(
                    "INSERT INTO metric_value (channel_id,node_id,code,value,unit,level,computed_at,"
                    "params_hash,rpm_source,stale) VALUES (:c,:n,:code,:v,'','unset',:t,'abc','',0)"
                ),
                payload,
            )
            values += len(payload)
        sess.commit()
    say(f"　造数据：4 通道 × 3 指标 × {rows_per_channel} 点 = {values} 行")
    series = timed(
        "metrics.list_values(CH2, v_RMS) 取全量",
        lambda: metric_store.list_values(sess, task.id, "CH2", "v_RMS"),
    )
    say(
        f"　　返回 {len(series)} 行，Python 对象估算 "
        f"{len(series) * 400 / 2**20:.1f} MiB（ValueRow≈400 B）"
    )
    payload = sum(len(r.computed_at) + 8 for r in series)
    say(f"　　plotly 需要序列化的点数据约 {payload / 2**20:.1f} MiB（时间串 + float）")
    say(
        "　→ 趋势页 app/pages/trends.py:191-239 一次 list_values 全量、直接 add_trace，"
        "没有分页、没有抽稀、没有时间窗筛选"
    )
    sess.close()


# ---------------------------------------------------------------- 4. SQLite 并发写
def probe_sqlite_concurrency() -> None:
    head("[F4] SQLite 并发写：两个 Streamlit 会话/tab 同时导入会不会锁库")
    path, engine, sess = fresh("perf-lock.db")
    task, ch_ids = build_tree(sess, "并发")
    batch_root = WORK / "batches-100"
    sess.close()

    with engine.connect() as conn:
        mode = conn.exec_driver_sql("PRAGMA journal_mode").scalar()
        busy = conn.exec_driver_sql("PRAGMA busy_timeout").scalar()
    say(
        f"　journal_mode={mode}　busy_timeout={busy}（SQLAlchemy 未设 connect_args，用 SQLite 默认 5 s）"
    )

    results: dict[str, str] = {}

    def worker(name: str, batch_index: int) -> None:
        engine_x = make_engine(path)
        session_x = make_session_factory(engine_x)()
        try:
            start = time.perf_counter()
            import_batch(
                session_x, task.id, batch_root / f"batch{batch_index:03d}", ch_ids, compute=True
            )
            results[name] = f"成功 {(time.perf_counter() - start) * 1000:.0f} ms"
        except Exception as exc:  # noqa: BLE001
            results[name] = f"{type(exc).__name__}: {str(exc).splitlines()[0]}"
        finally:
            session_x.close()
            engine_x.dispose()

    threads = [threading.Thread(target=worker, args=(f"tab{i}", i)) for i in range(4)]
    for thread in threads:
        thread.start()
    for thread in threads:
        thread.join()
    for name in sorted(results):
        say(f"　{name}: {results[name]}")
    with engine.connect() as conn:
        say(
            f"　导入后 dataset 行数：{conn.exec_driver_sql('SELECT COUNT(*) FROM dataset').scalar()}（4 个并发应为 4）"
        )
        codes = [r[0] for r in conn.exec_driver_sql("SELECT code FROM dataset ORDER BY id")]
        say(f"　数据集编号：{codes}")

    say("")
    say(
        "　补充：并发建数据集时 _next_code 用「本任务已有条数 + 1」（store/datasets.py:123-127），"
        "两个写者读到同一 count 会生成同一个 code；dataset.code 是全局 UNIQUE。"
    )
    say("　另测：两个独立 sqlite3 连接同时写（模拟 tab 双击导入）")
    raw_results = []

    def raw_worker(index: int) -> None:
        conn = sqlite3.connect(path, timeout=5.0)
        try:
            conn.execute("BEGIN IMMEDIATE")
            conn.execute(
                "INSERT INTO deletion_log (entity_type,entity_id,entity_name,mode,cascade_counts,operator,created_at)"
                " VALUES ('probe',?,?, 'soft','{}','review',datetime('now'))",
                (index, f"w{index}"),
            )
            time.sleep(0.4)
            conn.commit()
            raw_results.append(f"w{index} ok")
        except Exception as exc:  # noqa: BLE001
            conn.rollback()
            raw_results.append(f"w{index} {type(exc).__name__}: {exc}")
        finally:
            conn.close()

    raw_threads = [threading.Thread(target=raw_worker, args=(i,)) for i in range(4)]
    for thread in raw_threads:
        thread.start()
    for thread in raw_threads:
        thread.join()
    for item in sorted(raw_results):
        say(f"　　{item}")


def main() -> None:
    head(f"Lion-Monitor 总体复查 · 性能探测　{datetime.now():%Y-%m-%d %H:%M:%S}")
    say(f"　机器：{__import__('platform').platform()}　Python {sys.version.split()[0]}")
    probe_single_import()
    probe_many_batches(100)
    probe_long_series()
    probe_sqlite_concurrency()
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text("\n".join(lines) + "\n", encoding="utf-8")
    print(f"\n已写：{OUT}")


if __name__ == "__main__":
    main()
